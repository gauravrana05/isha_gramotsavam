  A. Volunteer Context Detection
  // src/lib/services/volunteerContext.ts
  export async function getVolunteerCurrentAssignment(volunteerId: string) {
    const assignment = await db.volunteerAssignment.findFirst({
      where: {
        volunteerId,
        event: { status: { in: ['active', 'registration_open'] } },
        deletedAt: null
      },
      include: {
        event: true,
        venueLevelMapping: {
          include: { venue: true }
        }
      }
    });

    return assignment;
  }

  export async function smartDetectEntity(
    assignment: VolunteerAssignment, 
    entityType: 'fixture' | 'match'
  ) {
    const now = new Date();

    if (entityType === 'match') {
      // Find current/recent matches at volunteer's venue
      const recentMatch = await db.match.findFirst({
        where: {
          venueLevelMappingId: assignment.venueLevelMappingId,
          eventId: assignment.eventId,
          // Within last 6 hours or next 2 hours
          OR: [
            { createdAt: { gte: subHours(now, 6) } },
            { createdAt: { lte: addHours(now, 2) } }
          ]
        },
        orderBy: { createdAt: 'desc' }
      });

      if (recentMatch) return recentMatch.id;
    }

    // Default to fixture
    const activeFixture = await db.fixture.findFirst({
      where: {
        venueLevelMappingId: assignment.venueLevelMappingId,
        eventId: assignment.eventId,
        status: 'active'
      }
    });

    return activeFixture?.id;
  }
